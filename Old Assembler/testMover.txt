loop:

// Jump if right button pressed.
MOVI 0x02, R0 
AND R15, R0
CMPI 0, R0
BNE doRight 


// Jump if left button pressed.
MOVI 0x04, R0
AND R15, R0
CMPI 0, R0
BNE doLeft

// Jump if jump button pressed.
MOVI 0x08, R0
AND R15, R0
CMPI 0, R0
BNE doJump

// Keep polling input if nothing pressed.
BUC loop


doLeft:
MOVI 0x01, R0
LSHI 15, R0
MOVI 0x00, R1
STOR R1, R0
LOAD R1, R0
MOVI 0x03, R0
LSHI 14, R0
MOVI 0x00, R1
STOR R1, R0
LOAD R1, R0
MOV R1, R3
BUC waitVsyncHigh

doRight:
MOVI 0x01, R0
LSHI 15, R0
MOVI 0x0f, R1
STOR R1, R0
LOAD R1, R0
MOV R1, R3
MOVI 0x03, R0
LSHI 14, R0
MOVI 0x0f, R1
STOR R1, R0
BUC waitVsyncHigh

doJump:
MOVI 0x01, R0
LSHI 15, R0
LOAD R1, R0
ADDI 0x1, R1
STOR R1, R0
MOVI 0x03, R0
LSHI 14, R0
MOVI 0x03, R3
LOAD R1, R0
ADDI 0x1, R1
STOR R1, R0
BUC waitVsyncHigh

// Wait for vsync to go high.
waitVsyncHigh:
MOVI 0x01, R2
AND R15, R2
CMPI 0, R2
BEQ waitVsyncHigh

// Wait for vsync to go low.
waitVsyncLow:
MOVI 0x01, R2
AND R15, R2
CMPI 0, R2
BNE waitVsyncLow 

BUC loop
