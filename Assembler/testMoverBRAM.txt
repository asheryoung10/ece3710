loop:

// -------- RIGHT BUTTON --------
MOVI 0x02, R0        // mask for right (bit 1)
AND R15, R0
CMPI 0, R0
BNE doRight          // if pressed → move right


// -------- LEFT BUTTON --------
MOVI 0x04, R0        // mask for left (bit 2)
AND R15, R0
CMPI 0, R0
BNE doLeft           // if pressed → go set to 0

BUC loop


// -------- LEFT: set position = 0 --------
doLeft:
MOVI 0x01, R0
LSHI 15, R0          // R0 = rect base addr
LOAD R1, R0          // get current position
SUBI 0x1, R1        // move left
STOR R1, R0

BUC waitVsyncHigh

// -------- RIGHT: increment position --------
doRight:
MOVI 0x01, R0
LSHI 15, R0          // R0 = rect base addr

LOAD R1, R0          // get current position
ADDI 0x1, R1        // move right
STOR R1, R0

BUC waitVsyncHigh


// -------- WAIT FOR VSYNC HIGH --------
waitVsyncHigh:
MOVI 0x01, R2        // mask for vsync (bit 0)
AND R15, R2
CMPI 0, R2
BEQ waitVsyncHigh    // stay until vsync == 1

// -------- WAIT FOR VSYNC LOW --------
waitVsyncLow:
MOVI 0x01, R2
AND R15, R2
CMPI 0, R2
BNE waitVsyncLow     // stay until vsync == 0
ADDI 0x1, R3

BUC loop
